4. DB TESTING COMMANDS
----------------------------------------------------------------------

-- View ALL records
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute('SELECT * FROM policy_acceptance')]; con.close()"

-- Count total accepted users
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); print('Total accepted:', con.execute('SELECT COUNT(*) FROM policy_acceptance').fetchone()[0]); con.close()"

-- Search by USERNAME (change jdoe to actual username)
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute(\"SELECT * FROM policy_acceptance WHERE username='TestUser'\")]; con.close()"

-- Search by PC NAME
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute(\"SELECT * FROM policy_acceptance WHERE pc_name='TestPC'\")]; con.close()"

-- Search by IP ADDRESS
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute(\"SELECT * FROM policy_acceptance WHERE ip_address='10.170.42.36'\")]; con.close()"

-- View records from TODAY only
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute(\"SELECT * FROM policy_acceptance WHERE DATE(start_time)=DATE('now')\")]; con.close()"

-- View LATEST 10 records
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute('SELECT * FROM policy_acceptance ORDER BY end_time DESC LIMIT 10')]; con.close()"

-- Show read duration in minutes and seconds
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(f\"User: {r[1]:20s} PC: {r[2]:15s} Duration: {r[7]//60}m {r[7]%60}s  IP: {r[4]}\") for r in con.execute('SELECT * FROM policy_acceptance ORDER BY end_time DESC')]; con.close()"

-- Check if SPECIFIC USER already accepted
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); r=con.execute(\"SELECT COUNT(*) FROM policy_acceptance WHERE username='TestUser'\").fetchone(); print('Accepted: YES' if r[0]>0 else 'Accepted: NO'); con.close()"

-- Delete/reset a specific user (force re-read)
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); con.execute(\"DELETE FROM policy_acceptance WHERE username='TestUser'\"); con.commit(); print('User reset done'); con.close()"

-- Delete ALL records (use for new policy version rollout)
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); con.execute('DELETE FROM policy_acceptance'); con.commit(); print('All records cleared'); con.close()"

-- Show DB table structure
python -c "import sqlite3; con=sqlite3.connect('C:/CBCTest/policy_log.db'); [print(r) for r in con.execute('PRAGMA table_info(policy_acceptance)')]; con.close()"

----------------------------------------------------------------------
6. DB COLUMN REFERENCE
----------------------------------------------------------------------

  Column 0  : id                  (auto number)
  Column 1  : username            (AD username)
  Column 2  : pc_name             (computer name)
  Column 3  : domain              (AD domain)
  Column 4  : ip_address          (real LAN IP)
  Column 5  : start_time          (kiosk opened time)
  Column 6  : end_time            (accept clicked time)
  Column 7  : read_duration_sec   (seconds spent reading)
  Column 8  : policy_version      (v1.0, v2.0 etc.)
  Column 9  : accepted            (1 = yes)
